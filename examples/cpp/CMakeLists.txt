# This directory also works as an independent installed-package consumer.
cmake_minimum_required(VERSION 4.2)
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RayTracingExample LANGUAGES CXX)
    option(RAYTRACING_BUILD_IO "Enable HDF5/CSV output" OFF)
    option(RAYTRACING_EXAMPLE_MATPLOT "Enable Matplot++" OFF)
    option(RAYTRACING_EXAMPLE_VTK "Enable VTK" OFF)
    if (RAYTRACING_BUILD_IO)
        enable_language(C)
        find_package(RayTracing CONFIG REQUIRED COMPONENTS IO)
    else ()
        find_package(RayTracing CONFIG REQUIRED COMPONENTS Core)
    endif ()
endif ()
add_executable(gaussian_beam_example gaussian_beam.cpp)
target_link_libraries(gaussian_beam_example PRIVATE RayTracing::Core)
add_executable(point_ray_example main.cpp)
target_link_libraries(point_ray_example PRIVATE RayTracing::Core)
if (RAYTRACING_BUILD_IO)
    target_link_libraries(point_ray_example PRIVATE RayTracing::IO)
    target_compile_definitions(point_ray_example PRIVATE RAYTRACING_EXAMPLE_IO)
endif ()
if (RAYTRACING_EXAMPLE_MATPLOT)
    find_package(Matplot++ CONFIG REQUIRED)
    target_sources(point_ray_example PRIVATE impulse_plot.cpp)
    target_link_libraries(point_ray_example PRIVATE Matplot++::matplot)
    target_compile_definitions(point_ray_example PRIVATE RAYTRACING_EXAMPLE_MATPLOT)
endif ()
if (RAYTRACING_EXAMPLE_VTK)
    find_package(VTK CONFIG REQUIRED COMPONENTS
            CommonCore CommonDataModel FiltersCore FiltersSources
            InteractionStyle RenderingCore RenderingOpenGL2 IOImage)
    target_sources(point_ray_example PRIVATE ray_view.cpp)
    target_link_libraries(point_ray_example PRIVATE ${VTK_LIBRARIES})
    target_compile_definitions(point_ray_example PRIVATE RAYTRACING_EXAMPLE_VTK)
    vtk_module_autoinit(TARGETS point_ray_example MODULES ${VTK_LIBRARIES})
endif ()
if (BUILD_TESTING)
    add_test(NAME PointRayExample COMMAND point_ray_example --rays 128)
    add_test(NAME GaussianBeamExample COMMAND gaussian_beam_example)
endif ()
