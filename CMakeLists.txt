cmake_minimum_required(VERSION 4.2)
if (POLICY CMP0207)
    cmake_policy(SET CMP0207 NEW)
endif ()
project(RayTracing VERSION 0.5.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(RAYTRACING_BUILD_PYTHON "Build the full Python API (including persistence)" OFF)
option(RAYTRACING_BUILD_IO "Build HDF5/CSV persistence" ${RAYTRACING_BUILD_PYTHON})
option(RAYTRACING_BUILD_EXAMPLES "Build the standalone C++ example" ${PROJECT_IS_TOP_LEVEL})
option(RAYTRACING_EXAMPLE_MATPLOT "Enable Matplot++ in the C++ example" OFF)
option(RAYTRACING_EXAMPLE_VTK "Enable VTK in the C++ example" OFF)
option(RAYTRACING_BUILD_BENCHMARKS "Build analytical spreading benchmarks" OFF)
if (PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif ()
if (RAYTRACING_BUILD_PYTHON AND NOT RAYTRACING_BUILD_IO)
    message(FATAL_ERROR "The Python API includes persistence; enable RAYTRACING_BUILD_IO.")
endif ()
add_subdirectory(RayTracing)
if (RAYTRACING_BUILD_PYTHON)
    add_subdirectory(python)
endif ()
if (RAYTRACING_BUILD_EXAMPLES)
    add_subdirectory(examples/cpp)
endif ()
if (BUILD_TESTING AND PROJECT_IS_TOP_LEVEL)
    foreach (test IN ITEMS SurfaceTests SequenceEvaluatorTests PathRefinementTests FieldResponseTests MaterialTests AntennaApiTests CylinderReturnTests BeamSolverTests)
        add_executable(${test} tests/${test}.cpp)
        target_link_libraries(${test} PRIVATE RayTracing::Core)
        add_test(NAME ${test} COMMAND ${test})
    endforeach ()
    if (RAYTRACING_BUILD_IO)
        foreach (test IN ITEMS AntennaApiTests CylinderReturnTests)
            target_link_libraries(${test} PRIVATE RayTracing::IO)
            target_compile_definitions(${test} PRIVATE RAYTRACING_TEST_IO)
        endforeach ()
    endif ()
    target_compile_definitions(AntennaApiTests PRIVATE TEST_PATTERN="${CMAKE_CURRENT_SOURCE_DIR}/antenna_patterns/simple_patch.ffs")
    target_compile_definitions(CylinderReturnTests PRIVATE TEST_PATTERN="${CMAKE_CURRENT_SOURCE_DIR}/antenna_patterns/simple_patch.ffs")
endif ()
if (RAYTRACING_BUILD_BENCHMARKS)
    add_executable(SpreadingBenchmarks benchmarks/SpreadingBenchmarks.cpp)
    target_link_libraries(SpreadingBenchmarks PRIVATE RayTracing::Core)
    if (BUILD_TESTING AND PROJECT_IS_TOP_LEVEL)
        add_test(NAME SpreadingBenchmarks COMMAND SpreadingBenchmarks ${CMAKE_CURRENT_BINARY_DIR}/benchmark-results)
    endif ()
endif ()
