cmake_minimum_required(VERSION 4.2)
if (POLICY CMP0207)
    cmake_policy(SET CMP0207 NEW)
endif ()
project(RayTracing VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
option(BUILD_BENCHMARKS "Build analytical spreading benchmarks" ON)
add_subdirectory(RayTracing)
add_subdirectory(python)
if (BUILD_TESTING)
    foreach (test IN ITEMS SurfaceTests SequenceEvaluatorTests PathRefinementTests FieldResponseTests MaterialTests AntennaApiTests CylinderReturnTests)
        add_executable(${test} tests/${test}.cpp)
        target_link_libraries(${test} PRIVATE RayTracing::IO)
        add_test(NAME ${test} COMMAND ${test})
    endforeach ()
    target_compile_definitions(AntennaApiTests PRIVATE TEST_PATTERN="${CMAKE_CURRENT_SOURCE_DIR}/antenna_patterns/simple_patch.ffs")
    target_compile_definitions(CylinderReturnTests PRIVATE TEST_PATTERN="${CMAKE_CURRENT_SOURCE_DIR}/antenna_patterns/simple_patch.ffs")
endif ()
if (BUILD_BENCHMARKS)
    add_executable(SpreadingBenchmarks benchmarks/SpreadingBenchmarks.cpp)
    target_link_libraries(SpreadingBenchmarks PRIVATE RayTracing::RayTracing)
    if (BUILD_TESTING)
        add_test(NAME SpreadingBenchmarks COMMAND SpreadingBenchmarks ${CMAKE_CURRENT_BINARY_DIR}/benchmark-results)
    endif ()
endif ()
